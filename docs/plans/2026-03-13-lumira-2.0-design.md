# Lumira 2.0 Design Document

**Date:** 2026-03-13
**Status:** Approved
**Target:** M4 Pro (48GB RAM), Hybrid local/cloud deployment

---

## Overview

Four major enhancements to transform Lumira into a more capable, reflective, and creative AI artist:

1. **FalkorDB Graph Memory** - Institutional memory with relationship tracking
2. **Inner Dialogue System** - Four inner voices for better decision-making
3. **Style Interpolation** - Mood blending, LoRA mixing, latent exploration
4. **FLUX.1 Schnell Previews** - Fast 4-step previews before full renders

### Deferred (Future Phases)

- Video generation (Kling/Runway)
- Community features (challenges, model sharing)
- 3D generation (TripoSR)
- Audio pairing (Suno)

## Phase 1: FalkorDB Graph Memory

### Purpose

Give Lumira institutional memory - she remembers not just *what* she created, but *why* decisions worked and how concepts relate.

### Graph Schema

```cypher
// Nodes
(:Mood {name: string, energy_range: [float, float], typical_styles: [string]})
(:Decision {id: string, prompt: string, timestamp: datetime, score: float})
(:Artwork {id: string, path: string, score: float, created_at: datetime})
(:Style {name: string, lora_path: string, description: string})

// Relationships
(:Mood)-[:INFLUENCES {strength: float}]->(:Decision)
(:Decision)-[:PRODUCED]->(:Artwork)
(:Decision)-[:USED {weight: float}]->(:Style)
(:Mood)-[:PAIRS_WELL_WITH {score: float}]->(:Mood)
(:Style)-[:BLENDS_WITH {score: float}]->(:Style)
(:Artwork)-[:SIMILAR_TO {distance: float}]->(:Artwork)
```

### Example Queries

```cypher
// What styles work when melancholic?
MATCH (m:Mood {name:'melancholic'})-[:INFLUENCES]->(d:Decision)-[:USED]->(s:Style)
WHERE d.score > 0.8
RETURN s.name, AVG(d.score) as avg_score
ORDER BY avg_score DESC

// What mood pairs produced the best work?
MATCH (m1:Mood)-[:INFLUENCES]->(d:Decision)<-[:INFLUENCES]-(m2:Mood)
WHERE d.score > 0.85 AND m1.name < m2.name
RETURN m1.name, m2.name, COUNT(*) as count, AVG(d.score) as avg
ORDER BY avg DESC

// Successful style combinations
MATCH (s1:Style)-[:BLENDS_WITH]->(s2:Style)
WHERE s1.name < s2.name
RETURN s1.name, s2.name, r.score
ORDER BY r.score DESC
```

### Integration

```python
# src/ai_artist/memory/graph_memory.py

from falkordb import FalkorDB

class GraphMemory:
    def __init__(self, host: str = "localhost", port: int = 6379):
        self.db = FalkorDB(host=host, port=port)
        self.graph = self.db.select_graph("lumira")

    async def record_decision(
        self,
        mood: str,
        prompt: str,
        styles: list[str],
        style_weights: list[float],
        outcome_score: float,
        artwork_path: str
    ) -> str:
        """Record a creative decision and its outcome."""
        decision_id = str(uuid4())

        query = """
        MERGE (m:Mood {name: $mood})
        CREATE (d:Decision {id: $decision_id, prompt: $prompt,
                           timestamp: datetime(), score: $score})
        CREATE (a:Artwork {id: $artwork_id, path: $artwork_path,
                          score: $score, created_at: datetime()})
        CREATE (m)-[:INFLUENCES]->(d)
        CREATE (d)-[:PRODUCED]->(a)
        WITH d
        UNWIND $styles AS style_data
        MERGE (s:Style {name: style_data.name})
        CREATE (d)-[:USED {weight: style_data.weight}]->(s)
        RETURN d.id
        """

        style_data = [{"name": s, "weight": w} for s, w in zip(styles, style_weights)]
        result = self.graph.query(query, params={...})
        return decision_id

    async def get_successful_styles_for_mood(
        self,
        mood: str,
        min_score: float = 0.75
    ) -> list[dict]:
        """Get styles that worked well for a given mood."""
        query = """
        MATCH (m:Mood {name: $mood})-[:INFLUENCES]->(d:Decision)-[:USED]->(s:Style)
        WHERE d.score >= $min_score
        RETURN s.name as style, AVG(d.score) as avg_score, COUNT(*) as uses
        ORDER BY avg_score DESC
        LIMIT 10
        """
        return self.graph.query(query, params={"mood": mood, "min_score": min_score})

    async def record_style_blend_success(
        self,
        style_a: str,
        style_b: str,
        score: float
    ):
        """Track which style combinations work well."""
        query = """
        MERGE (a:Style {name: $style_a})
        MERGE (b:Style {name: $style_b})
        MERGE (a)-[r:BLENDS_WITH]->(b)
        SET r.score = CASE
            WHEN r.score IS NULL THEN $score
            ELSE (r.score * r.count + $score) / (r.count + 1)
        END,
        r.count = COALESCE(r.count, 0) + 1
        """
        self.graph.query(query, params={...})
```

## Phase 2: Inner Dialogue System

### Purpose

Lumira consults four inner voices before creating, improving decision quality through structured reflection.

### The Four Voices

| Voice | Role | Implementation |
|-------|------|----------------|
| **The Dreamer** | Generates creative ideas, pushes boundaries | LLM prompt with high temperature |
| **The Critic** | Evaluates quality, spots weaknesses | Existing ArtistCritic, enhanced |
| **The Curator** | Ensures portfolio coherence | Graph queries for recent work patterns |
| **The Rememberer** | Surfaces relevant memories | FalkorDB + ChromaDB hybrid retrieval |

### Dialogue Flow

```python
# src/ai_artist/personality/dialogue.py

from dataclasses import dataclass
from enum import Enum

class Voice(Enum):
    DREAMER = "dreamer"
    CRITIC = "critic"
    CURATOR = "curator"
    REMEMBERER = "rememberer"

@dataclass
class DialogueTurn:
    voice: Voice
    message: str
    timestamp: datetime

@dataclass
class Concept:
    subject: str
    mood_blend: dict[str, float]  # {"melancholic": 0.7, "serene": 0.3}
    style_blend: dict[str, float]  # {"impressionist": 0.6, "cyberpunk": 0.4}
    prompt: str
    approved: bool = False

class InnerDialogue:
    MAX_ITERATIONS = 2

    def __init__(
        self,
        dreamer: Dreamer,
        critic: ArtistCritic,
        curator: Curator,
        rememberer: Rememberer,
        websocket: WebSocketManager
    ):
        self.dreamer = dreamer
        self.critic = critic
        self.curator = curator
        self.rememberer = rememberer
        self.ws = websocket
        self.history: list[DialogueTurn] = []

    async def deliberate(self, theme: str | None, mood: MoodState) -> Concept:
        """Run inner dialogue to develop a concept."""

        # The Rememberer provides context
        context = await self.rememberer.recall(mood=mood.name, theme=theme)
        await self._broadcast(Voice.REMEMBERER,
            f"I recall {len(context.similar_works)} similar works... "
            f"Best styles for this mood: {context.top_styles[:3]}")

        # The Dreamer proposes ideas
        ideas = await self.dreamer.imagine(context, mood, count=3)
        await self._broadcast(Voice.DREAMER,
            f"What if we explored: {ideas[0].subject}? "
            f"Or perhaps {ideas[1].subject}...")

        # The Curator filters for portfolio fit
        filtered = await self.curator.filter(ideas, context.recent_work)
        await self._broadcast(Voice.CURATOR,
            f"Given our recent work, {filtered[0].subject} would add "
            f"the most to our portfolio...")

        # The Critic evaluates (max 2 refinement loops)
        concept = filtered[0]
        for iteration in range(self.MAX_ITERATIONS):
            evaluation = await self.critic.evaluate(concept)
            await self._broadcast(Voice.CRITIC, evaluation.feedback)

            if evaluation.approved:
                concept.approved = True
                break

            # Dreamer refines based on feedback
            concept = await self.dreamer.refine(concept, evaluation.feedback)
            await self._broadcast(Voice.DREAMER,
                f"Let me reconsider... What about {concept.subject}?")

        return concept

    async def _broadcast(self, voice: Voice, message: str):
        """Broadcast dialogue turn to WebSocket clients."""
        turn = DialogueTurn(voice=voice, message=message, timestamp=datetime.now())
        self.history.append(turn)
        await self.ws.broadcast({
            "type": "inner_dialogue",
            "voice": voice.value,
            "message": message,
            "timestamp": turn.timestamp.isoformat()
        })
```

```python
# src/ai_artist/personality/inner_voices.py

class Dreamer:
    """Generates creative ideas with high temperature."""

    SYSTEM_PROMPT = """You are The Dreamer, Lumira's creative voice.
    You generate wild, boundary-pushing artistic ideas.
    Be imaginative, unexpected, and bold.
    Suggest specific subjects, moods, and style combinations."""

    async def imagine(
        self,
        context: MemoryContext,
        mood: MoodState,
        count: int = 3
    ) -> list[Concept]:
        prompt = f"""
        Current mood: {mood.name} (energy: {mood.energy})
        Recent themes to avoid: {context.recent_subjects}
        Successful styles for this mood: {context.top_styles}

        Generate {count} creative ideas for artwork.
        Push boundaries while respecting the mood.
        """

        response = await self.llm.generate(
            system=self.SYSTEM_PROMPT,
            prompt=prompt,
            temperature=0.9
        )
        return self._parse_ideas(response)

    async def refine(self, concept: Concept, feedback: str) -> Concept:
        """Refine a concept based on Critic's feedback."""
        prompt = f"""
        Original concept: {concept.subject}
        Feedback: {feedback}

        Refine this concept to address the feedback while
        keeping the creative spirit alive.
        """
        response = await self.llm.generate(
            system=self.SYSTEM_PROMPT,
            prompt=prompt,
            temperature=0.7
        )
        return self._parse_refined(response, concept)


class Curator:
    """Ensures portfolio coherence and variety."""

    def __init__(self, graph_memory: GraphMemory):
        self.graph = graph_memory

    async def filter(
        self,
        ideas: list[Concept],
        recent_work: list[Artwork]
    ) -> list[Concept]:
        """Filter ideas for portfolio fit."""
        recent_subjects = {w.subject for w in recent_work[-5:]}
        recent_styles = [w.style for w in recent_work[-5:]]

        scored = []
        for idea in ideas:
            score = 0

            # Novelty: not too similar to recent work
            if idea.subject not in recent_subjects:
                score += 2

            # Variety: different style from recent
            style_overlap = len(set(idea.style_blend.keys()) & set(recent_styles))
            score += (3 - style_overlap)

            # Coherence: fits overall artistic direction
            coherence = await self._check_coherence(idea)
            score += coherence

            scored.append((score, idea))

        return [idea for _, idea in sorted(scored, reverse=True)]


class Rememberer:
    """Surfaces relevant memories using hybrid retrieval."""

    def __init__(self, graph: GraphMemory, vectors: VectorMemory):
        self.graph = graph
        self.vectors = vectors

    async def recall(
        self,
        mood: str,
        theme: str | None
    ) -> MemoryContext:
        """Hybrid retrieval: graph for relationships, vectors for similarity."""

        # Graph: what worked for this mood?
        successful_styles = await self.graph.get_successful_styles_for_mood(mood)
        mood_pairs = await self.graph.get_mood_pairs(mood)

        # Vectors: similar past work
        if theme:
            similar = await self.vectors.search_creations(theme, limit=5)
        else:
            similar = []

        # Recent work for variety checking
        recent = await self.graph.get_recent_decisions(limit=10)

        return MemoryContext(
            top_styles=successful_styles,
            mood_pairs=mood_pairs,
            similar_works=similar,
            recent_work=recent,
            recent_subjects=[d.prompt.split()[0] for d in recent]
        )
```

### Mood Blending

```python
# src/ai_artist/core/mood_blender.py

@dataclass
class BlendedMood:
    components: dict[str, float]  # {"melancholic": 0.7, "serene": 0.3}
    energy: float
    colors: list[str]
    preferred_styles: list[str]
    name: str  # "melancholic+serene"

class MoodBlender:
    def __init__(self, mood_profiles: dict[str, MoodProfile]):
        self.profiles = mood_profiles

    def blend(self, weights: dict[str, float]) -> BlendedMood:
        """Blend multiple moods with given weights."""
        # Normalize weights
        total = sum(weights.values())
        weights = {k: v/total for k, v in weights.items()}

        # Weighted average of energy
        energy = sum(
            self.profiles[m].energy * w
            for m, w in weights.items()
        )

        # Blend color palettes
        colors = self._blend_palettes(weights)

        # Merge style preferences (union, weighted)
        styles = self._blend_styles(weights)

        return BlendedMood(
            components=weights,
            energy=energy,
            colors=colors,
            preferred_styles=styles,
            name="+".join(f"{m}({w:.0%})" for m, w in weights.items())
        )

    def _blend_palettes(self, weights: dict[str, float]) -> list[str]:
        """Blend color palettes from multiple moods."""
        all_colors = {}
        for mood, weight in weights.items():
            for color in self.profiles[mood].colors:
                all_colors[color] = all_colors.get(color, 0) + weight

        # Return top colors by weight
        sorted_colors = sorted(all_colors.items(), key=lambda x: -x[1])
        return [c for c, _ in sorted_colors[:5]]
```

```python
# src/ai_artist/core/style_interpolator.py

class StyleInterpolator:
    def __init__(self, pipe: DiffusionPipeline, lora_registry: dict[str, str]):
        self.pipe = pipe
        self.registry = lora_registry  # {"impressionist": "path/to/lora.safetensors"}
        self.loaded_loras: set[str] = set()

    async def load_styles(self, styles: list[str]):
        """Load required LoRAs (lazy loading)."""
        for style in styles:
            if style not in self.loaded_loras:
                path = self.registry[style]
                self.pipe.load_lora_weights(path, adapter_name=style)
                self.loaded_loras.add(style)

    async def set_blend(self, weights: dict[str, float]):
        """Set active style blend."""
        await self.load_styles(list(weights.keys()))

        self.pipe.set_adapters(
            adapter_names=list(weights.keys()),
            adapter_weights=list(weights.values())
        )

    async def generate_with_blend(
        self,
        prompt: str,
        style_weights: dict[str, float],
        **kwargs
    ) -> Image:
        """Generate image with blended styles."""
        await self.set_blend(style_weights)
        return self.pipe(prompt=prompt, **kwargs).images[0]
```

### Latent Space Exploration

```python
# src/ai_artist/core/style_interpolator.py (continued)

class LatentExplorer:
    def __init__(self, pipe: DiffusionPipeline):
        self.pipe = pipe
        self.text_encoder = pipe.text_encoder
        self.tokenizer = pipe.tokenizer

    def encode_prompt(self, prompt: str) -> torch.Tensor:
        """Encode prompt to embedding space."""
        tokens = self.tokenizer(
            prompt,
            padding="max_length",
            max_length=77,
            return_tensors="pt"
        )
        return self.text_encoder(tokens.input_ids)[0]

    def spherical_lerp(
        self,
        embed_a: torch.Tensor,
        embed_b: torch.Tensor,
        t: float
    ) -> torch.Tensor:
        """Spherical linear interpolation between embeddings."""
        # Normalize
        a_norm = embed_a / embed_a.norm(dim=-1, keepdim=True)
        b_norm = embed_b / embed_b.norm(dim=-1, keepdim=True)

        # Compute angle
        dot = (a_norm * b_norm).sum(dim=-1, keepdim=True).clamp(-1, 1)
        omega = torch.acos(dot)

        # Interpolate
        sin_omega = torch.sin(omega)
        s_a = torch.sin((1 - t) * omega) / sin_omega
        s_b = torch.sin(t * omega) / sin_omega

        return s_a * embed_a + s_b * embed_b

    async def explore_between(
        self,
        concept_a: str,
        concept_b: str,
        steps: int = 5
    ) -> list[Image]:
        """Generate images along path from A to B."""
        embed_a = self.encode_prompt(concept_a)
        embed_b = self.encode_prompt(concept_b)

        images = []
        for i in range(steps):
            t = i / (steps - 1)
            blended = self.spherical_lerp(embed_a, embed_b, t)

            image = self.pipe(
                prompt_embeds=blended,
                num_inference_steps=28
            ).images[0]
            images.append(image)

        return images
```

```python
# src/ai_artist/core/preview_generator.py

class PreviewGenerator:
    def __init__(self, device: str = "mps"):
        self.device = device
        self.schnell = None  # Lazy load
        self.style_interpolator = None

    async def ensure_loaded(self):
        """Lazy load FLUX.1 Schnell."""
        if self.schnell is None:
            from diffusers import FluxPipeline

            self.schnell = FluxPipeline.from_pretrained(
                "black-forest-labs/FLUX.1-schnell",
                torch_dtype=torch.bfloat16
            ).to(self.device)

            # Enable Metal optimizations
            self.schnell.enable_attention_slicing()

            self.style_interpolator = StyleInterpolator(
                self.schnell,
                load_lora_registry()
            )

    async def preview(
        self,
        prompt: str,
        style_weights: dict[str, float] | None = None,
        mood_blend: dict[str, float] | None = None
    ) -> tuple[Image, float]:
        """Generate fast preview. Returns (image, generation_time)."""
        await self.ensure_loaded()

        start = time.time()

        # Apply style blend if specified
        if style_weights:
            await self.style_interpolator.set_blend(style_weights)

        # Enhance prompt with mood colors if blending
        if mood_blend:
            blender = MoodBlender(load_mood_profiles())
            blended = blender.blend(mood_blend)
            prompt = f"{prompt}, {', '.join(blended.colors)} color palette"

        image = self.schnell(
            prompt=prompt,
            num_inference_steps=4,  # Schnell optimized for 4 steps
            height=1024,
            width=1024,
            guidance_scale=0.0  # Schnell doesn't use guidance
        ).images[0]

        elapsed = time.time() - start
        return image, elapsed

    async def preview_variations(
        self,
        prompt: str,
        style_weights: dict[str, float],
        variations: int = 4
    ) -> list[Image]:
        """Generate multiple preview variations."""
        images = []
        for i in range(variations):
            image, _ = await self.preview(
                prompt=prompt,
                style_weights=style_weights
            )
            images.append(image)
        return images
```

```python
        concept_b=request.concept_b,
        steps=request.steps
    )
    return ExploreResponse(
        images=[encode_base64(img) for img in images]
    )
```
